1. **src/data/firebase.ts** — Replace web SDK with native SDK:
   ```typescript
   import auth from '@react-native-firebase/auth';
   import firestore from '@react-native-firebase/firestore';

   // Native SDK initializes automatically from google-services.json / GoogleService-Info.plist
   // No initializeApp() call needed — the native SDK reads platform config at build time

   export const firebaseAuth = auth;
   export const db = firestore;

   /**
    * Configure Firebase Firestore persistence settings.
    * Call this once during app startup.
    */
   export const configureFirebase = async (): Promise<void> => {
     try {
       const settings = firestore().settings({
         persistence: true,
       });
       // Native SDK handles persistence automatically
       // This function kept for API compatibility with App.tsx
     } catch (err) {
       console.warn('Firestore configuration error:', err);
     }
   };

   /**
    * Clear Firestore persistence cache.
    * Useful for development or when schema changes.
    */
   export const clearFirebaseCache = async (): Promise<void> => {
     try {
       await firestore().clearPersistence();
     } catch (err) {
       console.error('Failed to clear Firestore cache:', err);
     }
   };
   ```
