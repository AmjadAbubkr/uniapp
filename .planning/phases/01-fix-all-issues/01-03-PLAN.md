1. **Create src/data/services/announcementService.ts** — Firestore CRUD for announcements:
   ```typescript
   import firestore from '@react-native-firebase/firestore';
   import { COLLECTIONS } from '../collections';
   import { LogService } from './logService';

   export interface AnnouncementData {
     title: string;
     content: string;
     type: 'subject' | 'faculty' | 'university';
     createdBy: string;
     createdAt?: any;
   }

   export const AnnouncementService = {
     create: async (data: AnnouncementData): Promise<{ success: boolean; id?: string; error?: string }> => {
       try {
         const docRef = await firestore().collection(COLLECTIONS.ANNOUNCEMENTS).add({
           ...data,
           createdAt: firestore.FieldValue.serverTimestamp(),
         });
         await LogService.logAction(data.createdBy, 'ANNOUNCEMENT_CREATED', docRef.id, {
           type: data.type,
           title: data.title,
         });
         return { success: true, id: docRef.id };
       } catch (error: any) {
         return { success: false, error: error?.message || 'Failed to create announcement' };
       }
     },

     getByType: async (type: string) => {
       const snap = await firestore().collection(COLLECTIONS.ANNOUNCEMENTS)
         .where('type', '==', type)
         .orderBy('createdAt', 'desc')
         .get();
       return snap.docs.map(doc => ({ id: doc.id, ...doc.data() }));
     },

     deleteAnnouncement: async (id: string, deletedBy: string): Promise<{ success: boolean; error?: string }> => {
       try {
         await firestore().collection(COLLECTIONS.ANNOUNCEMENTS).doc(id).delete();
         await LogService.logAction(deletedBy, 'ANNOUNCEMENT_DELETED', id, {});
         return { success: true };
       } catch (error: any) {
         return { success: false, error: error?.message || 'Failed to delete announcement' };
       }
     },
   };
   ```
